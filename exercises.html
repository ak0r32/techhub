  1. Takes an IP address as input
  2. Scans common ports (1-1024)
  3. Identifies open ports
  4. Reports service names for open ports

Educational use only. Never scan networks without permission. 

import socket

def scan_port(ip, port):
    """
    Check if a port is open on the given IP.
    """
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(1)
        result = sock.connect_ex((ip, port))
        sock.close()
        return result == 0
    except:
        return False
