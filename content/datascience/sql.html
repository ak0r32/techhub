What is SQL?

SQL (Structured Query Language) is the standard language for managing and manipulating relational databases. It's essential for data science, web development, and any role involving data.

Why Learn SQL?

  • Universal: Used by virtually all relational databases
  • Essential Skill: Required for data analyst, data scientist, backend developer roles
  • Efficient Data Retrieval: Query millions of records in seconds
  • Data Manipulation: Insert, update, delete data easily

Creating a Table

CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    email VARCHAR(100) UNIQUE,
    department VARCHAR(50),
    salary DECIMAL(10, 2),
    hire_date DATE
);

SELECT Queries

The SELECT statement is the most used SQL command for retrieving data.

Basic SELECT Examples

-- Select all columns
SELECT * FROM employees;

-- Select specific columns
SELECT first_name, last_name, salary FROM employees;

-- With WHERE clause
SELECT * FROM employees WHERE department = 'Engineering';

-- With ORDER BY
SELECT * FROM employees ORDER BY salary DESC;

-- With LIMIT
SELECT * FROM employees ORDER BY salary DESC LIMIT 5;

-- With WHERE and multiple conditions
SELECT * FROM employees 
WHERE department = 'Engineering' 
AND salary > 80000;

-- Using IN
SELECT * FROM employees 
WHERE department IN ('Engineering', 'Sales', 'Marketing');

-- Using LIKE for pattern matching
SELECT * FROM employees 
WHERE email LIKE '%@example.com';

Aggregate Functions

-- COUNT: Number of records
SELECT COUNT(*) FROM employees;

-- AVG: Average value
SELECT AVG(salary) FROM employees;

-- SUM: Total sum
SELECT SUM(salary) FROM employees;

-- MIN and MAX
SELECT MIN(salary), MAX(salary) FROM employees;

-- GROUP BY
SELECT department, AVG(salary) as avg_salary
FROM employees
GROUP BY department;

-- HAVING (filter groups)
SELECT department, COUNT(*) as emp_count
FROM employees
GROUP BY department
HAVING COUNT(*) > 10;

SQL JOINs

JOINs combine rows from two or more tables based on related columns.

Types of JOINs

-- INNER JOIN: Returns matching records from both tables
SELECT employees.first_name, employees.last_name, departments.dept_name
FROM employees
INNER JOIN departments ON employees.department_id = departments.dept_id;

-- LEFT JOIN: All records from left table, matching from right
SELECT employees.first_name, departments.dept_name
FROM employees
LEFT JOIN departments ON employees.department_id = departments.dept_id;

-- RIGHT JOIN: All records from right table, matching from left
SELECT employees.first_name, departments.dept_name
FROM employees
RIGHT JOIN departments ON employees.department_id = departments.dept_id;

-- FULL OUTER JOIN: All records from both tables
SELECT employees.first_name, departments.dept_name
FROM employees
FULL OUTER JOIN departments ON employees.department_id = departments.dept_id;

Advanced SQL

Subqueries

-- Subquery in WHERE clause
SELECT first_name, last_name, salary
FROM employees
WHERE salary > (SELECT AVG(salary) FROM employees);

-- Subquery with IN
SELECT first_name, last_name
FROM employees
WHERE department_id IN (
    SELECT dept_id FROM departments WHERE location = 'New York'
);

Window Functions

-- ROW_NUMBER: Assign unique row numbers
SELECT 
    first_name, 
    last_name, 
    salary,
    ROW_NUMBER() OVER (ORDER BY salary DESC) as rank
FROM employees;

-- RANK and DENSE_RANK
SELECT 
    first_name, 
    department,
    salary,
    RANK() OVER (PARTITION BY department ORDER BY salary DESC) as dept_rank
FROM employees;

-- Running totals
SELECT 
    first_name,
    salary,
    SUM(salary) OVER (ORDER BY hire_date) as running_total
FROM employees;

SQL with Python

Integrate SQL databases with Python for data analysis and applications.

Using SQLite with Python

import sqlite3
import pandas as pd

# Connect to database
conn = sqlite3.connect('employees.db')
cursor = conn.cursor()

# Execute query
cursor.execute('SELECT * FROM employees WHERE salary > 80000')
results = cursor.fetchall()

# Using pandas for easier data manipulation
df = pd.read_sql_query('SELECT * FROM employees', conn)
print(df.head())

# Insert data
cursor.execute('''
    INSERT INTO employees (first_name, last_name, salary)
    VALUES (?, ?, ?)
''', ('Alice', 'Smith', 90000))
conn.commit()

# Close connection
conn.close()

Using PostgreSQL with Python

import psycopg2
import pandas as pd

# Connect to PostgreSQL
conn = psycopg2.connect(
    host="localhost",
    database="mydb",
    user="username",
    password="password"
)

# Query and load into pandas
df = pd.read_sql_query('SELECT * FROM employees', conn)

# Perform data analysis
avg_salary = df['salary'].mean()
print(f"Average Salary: ${avg_salary:,.2f}")

conn.close()
